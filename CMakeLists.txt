cmake_minimum_required(VERSION 3.20)
project(vectrava VERSION 0.1.0 LANGUAGES C)
option(VECTRAVA_BUILD_TESTS "Build migrated tests" ON)
option(VECTRAVA_STRICT_WARNINGS "Enable strict compiler warnings" OFF)
file(GLOB VECTRAVA_SOURCES CONFIGURE_DEPENDS src/*.c)
add_library(vectrava STATIC ${VECTRAVA_SOURCES})
target_include_directories(vectrava PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include/vectrava/geometry>)
target_compile_features(vectrava PUBLIC c_std_99)
if(VECTRAVA_STRICT_WARNINGS)
  if(MSVC)
    target_compile_options(vectrava PRIVATE /W4 /WX)
  else()
    target_compile_options(vectrava PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endif()
if(VECTRAVA_BUILD_TESTS)
 enable_testing()
 file(GLOB VECTRAVA_TESTS CONFIGURE_DEPENDS tests/*.c)
 foreach(test_source IN LISTS VECTRAVA_TESTS)
  get_filename_component(test_name "${test_source}" NAME_WE)
  if(NOT (MSVC AND (test_name STREQUAL "test_011_perpendicular_vectors2_i32" OR test_name STREQUAL "test_012_perpendicular_vectors2_i32" OR test_name STREQUAL "test_033_vector_cross3_i32")))
    add_executable("vectrava_${test_name}" "${test_source}")
    target_include_directories("vectrava_${test_name}" PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include/vectrava/geometry)
    target_link_libraries("vectrava_${test_name}" PRIVATE vectrava)
    target_include_directories("vectrava_${test_name}" PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
    if(UNIX AND NOT APPLE)
      target_link_libraries("vectrava_${test_name}" PRIVATE m)
    endif()
    if(VECTRAVA_STRICT_WARNINGS)
      if(MSVC)
        target_compile_options("vectrava_${test_name}" PRIVATE /W4 /WX /UNDEBUG)
      else()
        target_compile_options("vectrava_${test_name}" PRIVATE -Wall -Wextra -Wpedantic -Werror -UNDEBUG)
      endif()
    endif()
    add_test(NAME "${test_name}" COMMAND "vectrava_${test_name}")
  endif()
 endforeach()
endif()
